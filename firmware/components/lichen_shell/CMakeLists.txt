# Release notes (#13): firmware/release_notes/<seq>-<date>.md are embedded as rodata; rn_files[] (release_notes.h) is a
# table of them generated here, so adding a note is just adding a file (CONFIGURE_DEPENDS re-runs this when one appears).
if(CMAKE_SCRIPT_MODE_FILE) # IDF's early requirements pass: CONFIGURE_DEPENDS is not allowed there
    file(GLOB RN_FILES ${CMAKE_CURRENT_SOURCE_DIR}/../../release_notes/*.md)
else()
    file(GLOB RN_FILES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/../../release_notes/*.md)
endif()
list(SORT RN_FILES)
set(RN_TABLE ${CMAKE_CURRENT_BINARY_DIR}/release_notes_table.c)
set(RN_C "// Generated by lichen_shell/CMakeLists.txt from firmware/release_notes/*.md (#13). Do not edit.\n")
string(APPEND RN_C [=[#include "release_notes.h"]=] "\n")
set(RN_ROWS "")
set(RN_N 0)
foreach(f ${RN_FILES})
    get_filename_component(RN_NAME ${f} NAME)
    string(APPEND RN_C "extern char const rn_note_${RN_N}[] asm(" [=["_binary_rn_note_]=] "${RN_N}" [=[_start");]=] "\n")
    string(APPEND RN_ROWS "    {" [=["]=] "${RN_NAME}" [=[", ]=] "rn_note_${RN_N}},\n")
    math(EXPR RN_N "${RN_N} + 1")
endforeach()
string(APPEND RN_C "rn_file_t const rn_files[] = {\n${RN_ROWS}    {0, 0}};\nsize_t const rn_file_count = ${RN_N};\n")
set(RN_OLD "")
if(EXISTS ${RN_TABLE})
    file(READ ${RN_TABLE} RN_OLD)
endif()
if(NOT "${RN_OLD}" STREQUAL "${RN_C}")
    file(WRITE ${RN_TABLE} "${RN_C}")
endif()
